\section{Shear (ASD)}

{\ttfamily\footnotesize\bfseries WHERE: TMS 402-16 \S 8.3.5 $\cdot$ ANSWERS: STEEL-ONLY / MASONRY-ONLY / COMBINED SHEAR CAPACITY $\cdot$ APPLIES: BEAMS, WALLS IN-PLANE, PIERS}

\begin{equation}
f_v = \frac{V}{A_{nv}} \le F_v = \left(F_{vm} + F_{vs}\right)\gamma_g
\end{equation}
{\ttfamily\footnotesize \begin{tabular}{@{}ll@{}}
$f_v$ & unit shear stress\\
$A_{nv}$ & net shear area\\
$F_v$ & total allowable unit shear stress\\
$F_{vm}$ & masonry share\\
$F_{vs}$ & shear-reinforcement share\\
$\gamma_g$ & 0.75 partially grouted shear wall, else 1.0\\
\end{tabular}}

\begin{equation}
F_{vm} = \frac{1}{2}\left[4.0 - 1.75\frac{M}{V d_v}\right]\sqrt{f'_m} + 0.25\frac{P}{A_{nv}}
\end{equation}
{\ttfamily\footnotesize \begin{tabular}{@{}ll@{}}
$d_v$ & actual member depth in direction of shear\\
$M/(Vd_v)$ & taken $\le 1.0$; axial term $P$ zero for beams\\
\end{tabular}}

\begin{equation}
F_{vs} = 0.5\,\frac{A_v F_s d_v}{A_{nv}\, s}
\end{equation}
{\ttfamily\footnotesize \begin{tabular}{@{}ll@{}}
$A_v$ & shear reinforcement area per spacing $s$\\
\end{tabular}}

\medskip{\ttfamily\footnotesize\bfseries LIMITS}

{\ttfamily\footnotesize \begin{tabular}{@{}ll@{}}
$M/(Vd_v) \le 0.25$ & $F_v \le 3\sqrt{f'_m}\,\gamma_g$\\
$M/(Vd_v) \ge 1.0$ & $F_v \le 2\sqrt{f'_m}\,\gamma_g$\\
between & interpolate\\
special reinforced shear wall & $F_{vm}$ coefficient $\frac{1}{2} \to \frac{1}{4}$; design for $1.5\times$ seismic $V$\\
\end{tabular}}

\section{Flexure (ASD)}

{\ttfamily\footnotesize\bfseries WHERE: TMS 402-16 \S 8.3.4 $\cdot$ ANSWERS: BEAM / WALL MOMENT CAPACITY $\cdot$ APPLIES: BEAMS, LINTELS, WALLS OUT-OF-PLANE}

\begin{equation}
n = \frac{E_s}{E_m} \qquad \rho = \frac{A_s}{bd} \qquad k = \sqrt{(n\rho)^2 + 2n\rho} - n\rho \qquad j = 1 - \frac{k}{3}
\end{equation}
{\ttfamily\footnotesize \begin{tabular}{@{}ll@{}}
$n$ & modular ratio\\
$k$ & neutral-axis depth ratio\\
$j$ & moment-arm ratio\\
\end{tabular}}

\begin{equation}
M \le \min\left( A_s F_s j d,\ \ \frac{1}{2} F_b k j b d^2 \right)
\end{equation}
{\ttfamily\footnotesize \begin{tabular}{@{}ll@{}}
$A_s F_s j d$ & steel governs\\
$\frac{1}{2}F_b kjbd^2$ & masonry governs\\
\end{tabular}}

\section{Axial + Interaction (ASD)}

{\ttfamily\footnotesize\bfseries WHERE: TMS 402-16 \S 8.2.4 (UNREINFORCED) $\cdot$ ANSWERS: UNREINFORCED WALL AXIAL CAPACITY, P+M UNITY CHECK}

\begin{equation}
F_a = \frac{1}{4}f'_m\left[1 - \left(\frac{h}{140r}\right)^2\right] \quad \left(\frac{h}{r} \le 99\right) \qquad
F_a = \frac{1}{4}f'_m\left(\frac{70r}{h}\right)^2 \quad \left(\frac{h}{r} > 99\right)
\end{equation}
{\ttfamily\footnotesize \begin{tabular}{@{}ll@{}}
$F_a$ & allowable axial compressive stress\\
$h$ & effective height\\
$r$ & radius of gyration\\
\end{tabular}}

\begin{equation}
\frac{f_a}{F_a} + \frac{f_b}{F_b} \le 1 \qquad\qquad P \le \frac{1}{4}P_e = \frac{1}{4}\,\frac{\pi^2 E_m I_n}{h^2}\left(1 - 0.577\frac{e}{r}\right)^3
\end{equation}
{\ttfamily\footnotesize \begin{tabular}{@{}ll@{}}
$f_a,\ f_b$ & computed axial, flexural stresses\\
$F_b$ here & $\frac{1}{3}f'_m$ unreinforced (\S 8.2.4.2), not $0.45f'_m$\\
$P_e$ & Euler buckling load; unreinforced check only\\
$I_n$ & net moment of inertia\\
$e$ & eccentricity\\
\end{tabular}}

\section{Shear + Flexure (Strength Design)}

{\ttfamily\footnotesize\bfseries WHERE: TMS 402-16 \S 9.3 $\cdot$ ANSWERS: $\phi V_n$ SHEAR / $\phi M_n$ MOMENT (SD) $\cdot$ $\phi$: SHEAR 0.80, FLEXURE 0.90 (\S 9.1.4)}

\begin{equation}
V_u \le \phi V_n = \phi\left(V_{nm} + V_{ns}\right)\gamma_g
\end{equation}
{\ttfamily\footnotesize \begin{tabular}{@{}ll@{}}
$V_{nm},\ V_{ns}$ & nominal masonry, reinforcement shares\\
\end{tabular}}

\begin{equation}
V_{nm} = \left[4.0 - 1.75\frac{M_u}{V_u d_v}\right]A_{nv}\sqrt{f'_m} + 0.25P_u
\qquad
V_{ns} = 0.5\,\frac{A_v}{s}f_y d_v
\end{equation}
{\ttfamily\footnotesize \begin{tabular}{@{}ll@{}}
$M_u/(V_u d_v)$ & taken $\le 1.0$\\
\end{tabular}}

\begin{equation}
a = \frac{A_s f_y + P_u}{0.80 f'_m b} \qquad M_n = \left(A_s f_y + P_u\right)\left(d - \frac{a}{2}\right)
\end{equation}
{\ttfamily\footnotesize \begin{tabular}{@{}ll@{}}
$a$ & equivalent stress-block depth ($0.80 f'_m$, \S 9.3.2)\\
$\varepsilon_{mu}$ & 0.0025 CMU, 0.0035 clay (\S 9.3.2)\\
\end{tabular}}

\medskip{\ttfamily\footnotesize\bfseries LIMITS}

{\ttfamily\footnotesize \begin{tabular}{@{}ll@{}}
$M_u/(V_u d_v) \le 0.25$ & $V_n \le 6 A_{nv}\sqrt{f'_m}\,\gamma_g$\\
$M_u/(V_u d_v) \ge 1.0$ & $V_n \le 4 A_{nv}\sqrt{f'_m}\,\gamma_g$\\
between & interpolate\\
special wall capacity design & shear $\ge$ shear at $1.25 M_n$, need not exceed $2.5 V_u$ (\S 7.3.2.6.1)\\
\end{tabular}}

\section{Anchor Bolts (ASD)}

{\ttfamily\footnotesize\bfseries WHERE: TMS 402-16 \S 8.1.3 $\cdot$ ANSWERS: HEADED-BOLT TENSION / SHEAR / COMBINED CAPACITY $\cdot$ MINIMUM MODE GOVERNS $\cdot$ BENT-BAR: ADD PULLOUT (EQ 8-3)}

\begin{equation}
B_a = \min\left( B_{ab} = 1.25 A_{pt}\sqrt{f'_m},\ \ B_{as} = 0.6 A_b f_y \right)
\end{equation}
{\ttfamily\footnotesize \begin{tabular}{@{}ll@{}}
$B_a$ & allowable tension\\
$B_{ab}$ & masonry breakout\\
$B_{as}$ & steel yield\\
$A_{pt}$ & projected tension breakout area\\
$A_b$ & bolt cross-sectional area\\
\end{tabular}}

\begin{equation}
B_v = \min\left( B_{vb} = 1.25 A_{pv}\sqrt{f'_m},\ \ B_{vc} = 350\sqrt[4]{f'_m A_b},\ \ B_{vpry} = 2.0 B_{ab},\ \ B_{vs} = 0.36 A_b f_y \right)
\end{equation}
{\ttfamily\footnotesize \begin{tabular}{@{}ll@{}}
$B_v$ & allowable shear\\
$B_{vb}$ & masonry breakout\\
$B_{vc}$ & masonry crushing\\
$B_{vpry}$ & pryout\\
$B_{vs}$ & steel yield\\
$A_{pv}$ & projected shear breakout area\\
\end{tabular}}

\begin{equation}
\frac{b_a}{B_a} + \frac{b_v}{B_v} \le 1
\end{equation}
{\ttfamily\footnotesize \begin{tabular}{@{}ll@{}}
$b_a,\ b_v$ & applied tension, shear\\
\end{tabular}}

\section{Worked Examples}

{\ttfamily\footnotesize\bfseries EX 1 $\cdot$ ASD SHEAR, MASONRY ONLY (\S 8.3.5.1.2) $\cdot$ $f'_m$ 2000, $A_{nv}$ 732 IN$^2$, $P$ 40 K, $M/(Vd_v)$ 1.0}
\begin{align*}
F_{vm} &= \tfrac{1}{2}\left[4.0 - 1.75(1.0)\right]\sqrt{2000} + 0.25\tfrac{40{,}000}{732} = 50.3 + 13.7 = 64.0\ \text{psi}\\
F_v &\le 2\sqrt{f'_m}\,\gamma_g = 89.4\ \text{psi}\ (\gamma_g = 1.0) \ \text{— OK} \qquad V = 64.0(732) = 46.8 \approx 47\ \text{kips}
\end{align*}
{\ttfamily\footnotesize TRAPS: drop axial term $\to$ 37 $\cdot$ axial inside the $\frac{1}{2}$ bracket $\to$ 42}

\medskip
{\ttfamily\footnotesize\bfseries EX 2 $\cdot$ ASD FLEXURE (\S 8.3.4) $\cdot$ $f'_m$ 1500, $f_y$ 60 KSI, $b$ 7.625, $d$ 20, $A_s$ 0.88 IN$^2$}
\begin{align*}
E_m &= 900f'_m;\quad n = \tfrac{29{,}000}{1350} = 21.5;\quad \rho = \tfrac{0.88}{7.625(20)} = 0.00577;\quad n\rho = 0.124\\
k &= \sqrt{(n\rho)^2 + 2n\rho} - n\rho = 0.389;\qquad j = 1 - k/3 = 0.870\\
M_s &= A_s F_s j d = 40.8\ \text{k-ft};\qquad M_m = \tfrac{1}{2}F_b k j b d^2 = 29.1\ \text{k-ft} \ \Rightarrow\ \text{masonry governs, } 29
\end{align*}
{\ttfamily\footnotesize TRAPS: report steel value $\to$ 41 $\cdot$ average of the two $\to$ 35}

\medskip
{\ttfamily\footnotesize\bfseries EX 3 $\cdot$ UNREINFORCED UNITY (\S 8.2.4) $\cdot$ $f'_m$ 1500, $h$ 12 FT, $r$ 2.2, $f_a$ 130, $f_b$ 200 PSI}
\begin{align*}
h/r &= 65.5 \le 99 \ \Rightarrow\ F_a = \tfrac{1}{4}f'_m\left[1 - \left(\tfrac{144}{308}\right)^2\right] = 293\ \text{psi (Eq 8-11)}\\
\tfrac{f_a}{F_a} + \tfrac{f_b}{F_b} &= \tfrac{130}{293} + \tfrac{200}{500} = 0.44 + 0.40 = 0.84 \qquad F_b = \tfrac{1}{3}f'_m = 500\ \text{psi, NOT } 0.45f'_m
\end{align*}
{\ttfamily\footnotesize TRAPS: reinforced $F_b$ 675 $\to$ 0.74 $\cdot$ wrong slenderness branch $\to$ 0.70 $\cdot$ axial alone $\to$ 0.44}

\medskip
{\ttfamily\footnotesize\bfseries EX 4 $\cdot$ SD SHEAR, SPECIAL WALL (\S 9.3.4.1.2) $\cdot$ $f'_m$ 3000, $A_{nv}$ 549 IN$^2$, $P_u$ 10 K, \#4 @ 16 IN}
\begin{align*}
M_u/(V_u d_v) &= 16/6 = 2.67 \to \text{take } 1.0\\
V_{nm} &= 2.25(549)\sqrt{3000} + 0.25(10{,}000) = 70.2\ \text{k};\qquad V_{ns} = 0.5\tfrac{0.20}{16}(60{,}000)(72) = 27.0\ \text{k}\\
V_n &= 97.2\ \text{k} \le 4A_{nv}\sqrt{f'_m} = 120.3\ \text{k — OK};\qquad \phi V_n = 0.80(97.2) = 77.7 \approx 78\ \text{kips}
\end{align*}
{\ttfamily\footnotesize TRAPS: $\phi V_{nm}$ only $\to$ 56 $\cdot$ $V_{nm}$ alone $\to$ 70 $\cdot$ forgot $\phi$ $\to$ 97}

\medskip
{\ttfamily\footnotesize\bfseries EX 5 $\cdot$ SD FLEXURE (\S 9.3.2) $\cdot$ $f'_m$ 2000, $f_y$ 60 KSI, $b$ 7.625, $d$ 20, $A_s$ 0.88 IN$^2$}
\begin{align*}
a &= \frac{A_s f_y}{0.80 f'_m b} = \frac{52{,}800}{12{,}200} = 4.33\ \text{in.}\qquad M_n = A_s f_y\left(d - \tfrac{a}{2}\right) = 78.5\ \text{k-ft}\\
\phi M_n &= 0.90(78.5) = 70.6 \approx 71\ \text{k-ft} \quad (\phi = 0.90\ \text{flexure, \S 9.1.4.1})
\end{align*}
{\ttfamily\footnotesize TRAPS: forgot $\phi$ $\to$ 79 $\cdot$ lever arm $d-a$ $\to$ 62 $\cdot$ $A_s f_y d$ alone $\to$ 88}
