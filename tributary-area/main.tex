\newcommand{\logicBlock}[1]{%
{\raggedright\scriptsize\hyphenpenalty=10000\exhyphenpenalty=10000 #1\par}%
}

\newcommand{\logicStress}{%
\logicBlock{%
Let \(\Omega\) be the slab plan and \(B\) the support faces.
Stress lines connect support-to-support only:
\[
L=\{\ell_k:\partial\ell_k\subset\partial B\}
\]
Endpoints are wall/column edges or corners.%
}%
}

\newcommand{\logicCentroid}{%
\logicBlock{%
Stress-line cells:
\[
\{P_i\}=\operatorname{faces}\bigl(\Omega\setminus(B\cup L)\bigr)
\]
Cell centroid:
\[
A_i=\int_{P_i}dA,\quad
c_i=\frac{1}{A_i}\int_{P_i}(x,y)\,dA
\]%
}%
}

\newcommand{\logicEdgePoint}{%
\logicBlock{%
Boundary node on an ordinary exposed interval:
\[
e_j=\gamma_j(\ell_j/2),\quad
\ell_j=\int_{\gamma_j}ds
\]
If \(\gamma_j\) crosses a slab corner, use the corner itself as
\(e_j\). Do not wrap tributary logic around slab corners.%
}%
}

\newcommand{\logicArea}{%
\logicBlock{%
Use stress-line midpoints \(s_k\), centroids \(c_i\), and boundary
nodes \(e_j\):
\[
G=\{c_i\leftrightarrow s_k,\;c_i\leftrightarrow e_j\}
\]
Tributary areas:
\[
\{T_m\}=\operatorname{faces}\bigl(\Omega\setminus(B\cup G)\bigr)
\]%
}%
}
