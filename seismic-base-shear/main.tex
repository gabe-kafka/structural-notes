\section{Effective Seismic Weight $W$}

\begin{equation}
W = D + W_{\text{part}} + W_{\text{equip}} + 0.25L_{\text{storage}} + W_{\text{snow,req}}
\end{equation}

\section{Seismic Response Coefficient $C_s$}

\subsection{Dependencies}

\begin{equation}
\begin{aligned}
S_{MS} &= F_aS_S, &
S_{M1} &= F_vS_1 \\
S_{DS} &= \frac{2}{3}S_{MS}, &
S_{D1} &= \frac{2}{3}S_{M1}
\end{aligned}
\end{equation}

\begin{equation}
T_a = C_th_n^x
\end{equation}

\begin{equation}
T =
\begin{cases}
T_a & \text{hand calc / approximate period} \\
\min(T_{\text{calc}}, C_uT_a) & \text{calculated period available}
\end{cases}
\end{equation}

\subsection{Coefficient Checks}

\begin{equation}
C_{s,0} = \frac{S_{DS}}{R/I_e}
\end{equation}

\begin{equation}
C_{s,\max} =
\begin{cases}
\dfrac{S_{D1}}{T(R/I_e)} & T \leq T_L \\[6pt]
\dfrac{S_{D1}T_L}{T^2(R/I_e)} & T > T_L
\end{cases}
\end{equation}

\begin{equation}
C_{s,S_1} =
\begin{cases}
\dfrac{0.5S_1}{R/I_e} & S_1 \geq 0.6g \\[6pt]
0 & S_1 < 0.6g
\end{cases}
\end{equation}

\begin{equation}
C_{s,\min} = \max\!\left(0.044S_{DS}I_e,\ 0.01,\ C_{s,S_1}\right)
\end{equation}

\begin{equation}
C_s = \max\!\left(C_{s,\min},\ \min(C_{s,0}, C_{s,\max})\right)
\end{equation}

{\ttfamily\footnotesize
CONTROL: use the highest controlling $C_s$; cap $C_{s,0}$ at $C_{s,\max}$, then compare to $C_{s,\min}$.
}

\section{Seismic Base Shear $V$}

ASCE 7 Eq. 12.8-1:
\begin{equation}
V = C_s W
\end{equation}
